# Arstro Ntwb - the C++ app side of NTWB, the native-to-web bridge (docs: README.md).
#
# Platform-free core (Json, Protocol, Client) + one transport seam (Transport.cpp holds the
# only OS calls). No dependency beyond the C++17 standard library, so any Arstro app -
# desktop, headless, later Android/WASM - can link it.
cmake_minimum_required(VERSION 3.16)
project(arstro_ntwb LANGUAGES CXX)

add_library(arstro_ntwb STATIC
  src/Json.cpp
  src/Protocol.cpp
  src/Transport.cpp
  src/Client.cpp)
target_include_directories(arstro_ntwb PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(arstro_ntwb PUBLIC cxx_std_17)

# ntwb_tests: JSON, framing, validation, the client over an in-memory transport, and - the
# one that keeps the protocol single-sourced - this library's message table against the
# vendored spec/api.json generated by Arstro Remote (NR-2).
add_executable(ntwb_tests tests/ntwbTests.cpp)
target_link_libraries(ntwb_tests PRIVATE arstro_ntwb)
target_compile_definitions(ntwb_tests PRIVATE NTWB_SPEC_JSON="${CMAKE_CURRENT_SOURCE_DIR}/spec/api.json")
add_test(NAME ntwb_tests COMMAND ntwb_tests)
